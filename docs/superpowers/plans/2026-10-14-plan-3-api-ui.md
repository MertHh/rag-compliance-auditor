**Goal:** Çekirdek kütüphaneyi (yükleme, arama, belgeye sor, denetim, iz, inceleme) bir FastAPI servisi olarak sunmak ve bu servisi çağıran altı sayfalık bir Streamlit arayüzü kurmak; alıntılar PDF üzerinde işaretlenir, denetim raporu JSON ve PDF olarak dışa aktarılır.

import threading
import time
from concurrent.futures import ThreadPoolExecutor

from fakes import CONTRACT_PAGES, FakeEmbedder, FakeReranker, FakeSparseEncoder

from auditor.index.qdrant_store import QdrantStore
from auditor.ingestion.indexer import Indexer
from auditor.retrieval.pipeline import Retriever
from auditor.storage.docstore import DocumentStore


class OverlapDetector:
    """Wraps an object and counts calls that start while another call is still running."""

    def __init__(self, inner):
        self._inner = inner
        self._guard = threading.Lock()
        self._active = 0
        self.overlaps = 0
        self.calls = 0

    def __getattr__(self, name):
        attr = getattr(self._inner, name)
        if not callable(attr):
            return attr

        def wrapped(*args, **kwargs):
            with self._guard:
                self._active += 1
                self.calls += 1
                if self._active > 1:
                    self.overlaps += 1
            try:
                time.sleep(0.002)
                return attr(*args, **kwargs)
            finally:
                with self._guard:
                    self._active -= 1

        return wrapped


def test_searches_and_ingest_never_use_the_store_at_once(tmp_path, make_pdf, sample_contract):
    store = OverlapDetector(QdrantStore.in_memory())
    docstore = DocumentStore(tmp_path / "docs")
    embedder, sparse = FakeEmbedder(), FakeSparseEncoder()
    lock = threading.Lock()
    indexer = Indexer(store, embedder, sparse, docstore, lock=lock)
    retriever = Retriever(store, embedder, sparse, FakeReranker(), docstore, lock)
    indexer.ingest_pdf(make_pdf(tmp_path / "a.pdf", CONTRACT_PAGES))

    with ThreadPoolExecutor(max_workers=8) as pool:
        searches = [pool.submit(retriever.search, "total liability fees") for _ in range(12)]
        ingest = pool.submit(indexer.ingest_text, sample_contract, "b.txt")
        results = [future.result() for future in searches]
        assert ingest.result().chunk_count > 0

    assert store.overlaps == 0
    assert store.calls > 13
    assert all(result.chunks for result in results)


def test_retriever_serializes_its_own_searches(tmp_path, make_pdf):
    store = OverlapDetector(QdrantStore.in_memory())
    docstore = DocumentStore(tmp_path / "docs")
    embedder, sparse = FakeEmbedder(), FakeSparseEncoder()

    def __init__(self, store: VectorStore, embedder: Embedder, sparse_encoder: SparseEncoder,
                 reranker: Reranker, docstore: DocumentStore, lock: threading.Lock | None = None):
        self._store = store
        self._embedder = embedder
        self._sparse = sparse_encoder
        self._reranker = reranker
        self._docstore = docstore

        self._lock = lock or threading.Lock()

                flt: SearchFilter | None) -> RetrievalResult:
        if not query.strip():
            return RetrievalResult(query, config, [], [], True)

    def is_indexed(self, document_id: str) -> bool:
        with self._lock:
            return self._docstore.exists(document_id) and self._store.has_document(document_id)

    def _index(self, doc: ParsedDocument, metadata: dict | None, source_pdf: bytes | None,
               extract_metadata: MetadataFn | None) -> IngestResult:
        if self.is_indexed(doc.document_id):
            return IngestResult(doc.document_id, 0, True)

        chunks = build_chunks(
            doc, detect_sections(doc.lines), self._max_child_tokens, self._max_parent_tokens
        )
        if not chunks:
            raise EmptyDocumentError(f"{doc.filename}: document has no text to index")

        # A field the caller left empty (None) must not erase an extracted value.
        given = {key: value for key, value in (metadata or {}).items() if value is not None}
        meta = {
            "filename": doc.filename,
            "doc_type": None,
            "date": None,
            "parties": [],
            "jurisdiction": None,
            **(extract_metadata(doc) if extract_metadata else {}),
            **given,
        }
        if self._contextualizer:
            for chunk in chunks:
                chunk.context_prefix = self._contextualizer(doc, chunk)
        texts = [
            f"{chunk.context_prefix}\n\n{chunk.text}" if chunk.context_prefix else chunk.text
            for chunk in chunks
        ]

        # Save the text before the vectors: a search hit must always have a
        # document behind it. A failed upsert is retried by the next ingest.
        # Metadata extraction (an LLM call) stays outside the lock.
        with self._lock:
            self._docstore.save(doc, meta, source_pdf)
            self._store.ensure_collection(self._embedder.dim)
            self._store.delete_document(doc.document_id)
            self._store.upsert(
                chunks,
                self._embedder.embed_documents(texts),
                self._sparse.encode_documents(texts),
                meta,
            )
        return IngestResult(doc.document_id, len(chunks), False)
```

import threading
import time
from concurrent.futures import ThreadPoolExecutor

from auditor.lazy import Lazy


class Model:
    dim = 64

    def encode(self, text):
        return text.upper()


def test_builds_on_first_attribute_access_only():
    built = []
    lazy = Lazy(lambda: built.append(1) or Model())
    assert built == [] and lazy.loaded is False
    assert lazy.dim == 64
    assert lazy.encode("a") == "A"
    assert built == [1] and lazy.loaded is True


def test_concurrent_first_use_builds_once():
    built = []
    lock = threading.Lock()

    def factory():
        time.sleep(0.05)
        with lock:
            built.append(1)
        return Model()

    lazy = Lazy(factory)
    with ThreadPoolExecutor(max_workers=8) as pool:
        assert list(pool.map(lambda _: lazy.dim, range(8))) == [64] * 8
    assert built == [1]


def test_failed_build_is_retried_on_next_use():
    attempts = []

    def factory():
        attempts.append(1)
        if len(attempts) == 1:
            raise OSError("download failed")
        return Model()

    lazy = Lazy(factory)
    try:
        lazy.load()
    except OSError:
        pass
    assert lazy.dim == 64
    assert len(attempts) == 2

```python
import threading
from typing import Callable, Generic, TypeVar

T = TypeVar("T")


class Lazy(Generic[T]):
    """Stands in for an object that is built on first use. Loading the embedding and
    reranker models takes tens of seconds; work that never searches, such as a cached
    audit, skips it. A failed build is retried on the next use."""

    def __init__(self, factory: Callable[[], T]):
        self._factory = factory
        self._value: T | None = None
        self._build_lock = threading.Lock()

    @property
    def loaded(self) -> bool:
        return self._value is not None

    def load(self) -> T:
        if self._value is None:
            with self._build_lock:
                if self._value is None:
                    self._value = self._factory()
        return self._value

    def __getattr__(self, name: str):
        return getattr(self.load(), name)
```

def cache_key(document_id: str, playbook_sha256: str, provider_name: str, model_label: str,
              prompt_version: str = PROMPT_VERSION) -> str:
    raw = "\n".join([document_id, playbook_sha256, provider_name, model_label, prompt_version])
    return hashlib.sha256(raw.encode("utf-8")).hexdigest()


@dataclass(frozen=True)
class PreparedAudit:
    document_id: str
    playbook: LoadedPlaybook
    provider: LLMProvider
    cache_key: str


class Auditor:
    def __init__(self, retriever: Retriever, docstore: DocumentStore, audits: AuditStore,
                 playbooks_dir: str | Path, provider_factory: Callable[[str], LLMProvider],
                 config: AuditConfig = AuditConfig()):
        self._retriever = retriever
        self._docstore = docstore
        self._audits = audits
        self._playbooks_dir = Path(playbooks_dir)
        self._providers = provider_factory
        self._config = config

    def prepare(self, document_id: str, playbook_key: str, provider_name: str) -> PreparedAudit:
        """Checks everything that can fail before any model or LLM work starts."""
        if not self._docstore.exists(document_id):
            raise DocumentNotFound(document_id)
        loaded = find_playbook(self._playbooks_dir, playbook_key)
        provider = self._providers(provider_name)
        provider.check_available()
        key = cache_key(document_id, loaded.sha256, provider.name, models_label(provider))
        return PreparedAudit(document_id, loaded, provider, key)

    def cached(self, prepared: PreparedAudit) -> dict | None:
        report = self._audits.find_cached(prepared.cache_key)
        return None if report is None else {**report, "cached": True}

    def execute(self, prepared: PreparedAudit, audit_id: str | None = None) -> dict:
        recorder = TraceRecorder()
        nodes = RuleNodes(self._retriever, self._docstore, prepared.provider, recorder,
                          prepared.document_id, self._config)
        graph = build_rule_graph(nodes, self._config.max_rewrites)
        rules = prepared.playbook.playbook.rules
        started = time.perf_counter()
        outcomes = graph.batch(
            [initial_state(rule) for rule in rules],
            config={"max_concurrency": self._config.concurrency},
            return_exceptions=True,
        )
        findings = [build_finding(rule, outcome) for rule, outcome in zip(rules, outcomes, strict=True)]
        report = build_report(
            audit_id=audit_id or uuid.uuid4().hex,
            document_id=prepared.document_id,
            playbook_key=prepared.playbook.key,
            provider=prepared.provider,
            findings=findings,
            duration_s=time.perf_counter() - started,
            usage=recorder.totals(),
        )
        self._audits.save_audit(report, recorder.to_dicts(), prepared.cache_key)
        return report

    def run(self, document_id: str, playbook_key: str, provider_name: str, *,
            force: bool = False) -> dict:
        prepared = self.prepare(document_id, playbook_key, provider_name)
        if not force:
            report = self.cached(prepared)
            if report is not None:
                return report
        return self.execute(prepared)

import logging
import threading
from contextlib import asynccontextmanager

from dotenv import load_dotenv
from fastapi import FastAPI, Request
from fastapi.responses import JSONResponse

from auditor.api.routes import meta
from auditor.audit.playbook import PlaybookError, PlaybookNotFound
from auditor.config import Settings
from auditor.index.store import InvalidDateError
from auditor.ingestion.indexer import EmptyDocumentError
from auditor.ingestion.parser import InvalidPdfError, ScannedPdfError
from auditor.llm.base import LLMError, ProviderUnavailableError
from auditor.services import Services, build_services
from auditor.storage.db import AuditNotFound, FindingNotFound
from auditor.storage.docstore import DocumentNotFound

log = logging.getLogger(__name__)

STATUS_CODES: dict[type[Exception], int] = {
    DocumentNotFound: 404,
    AuditNotFound: 404,
    FindingNotFound: 404,
    PlaybookNotFound: 404,
    PlaybookError: 422,
    InvalidPdfError: 422,
    ScannedPdfError: 422,
    EmptyDocumentError: 422,
    InvalidDateError: 422,
    ProviderUnavailableError: 503,
    LLMError: 502,
}
_NOT_FOUND_NOUNS = {DocumentNotFound: "document", AuditNotFound: "audit", FindingNotFound: "finding"}


def error_detail(exc: Exception) -> str:
    noun = _NOT_FOUND_NOUNS.get(type(exc))
    if noun is not None:
        return f"{noun} {exc.args[0] if exc.args else ''} not found"
    return str(exc)


def _status_for(exc: Exception) -> int:
    return next(STATUS_CODES[cls] for cls in type(exc).__mro__ if cls in STATUS_CODES)


async def _handle_core_error(request: Request, exc: Exception) -> JSONResponse:
    return JSONResponse({"detail": error_detail(exc)}, status_code=_status_for(exc))


def _warm_up(services: Services) -> None:
    try:
        services.warm_up()
    except Exception:
        log.exception("model warm-up failed; models will load on first use")


def create_app(services: Services | None = None) -> FastAPI:
    @asynccontextmanager
    async def lifespan(app: FastAPI):
        if services is None:
            load_dotenv()
            app.state.services = build_services(Settings.from_env())
            threading.Thread(target=_warm_up, args=(app.state.services,), daemon=True).start()
        else:
            app.state.services = services
        yield

    app = FastAPI(title="Compliance Auditor API", lifespan=lifespan)
    for exc_type in STATUS_CODES:
        app.add_exception_handler(exc_type, _handle_core_error)
    app.include_router(meta.router)
    return app
```

MAX_UPLOAD_BYTES = 50 * 1024 * 1024
_UNSAFE_CHARS = re.compile(r'[<>:"/\\|?*\x00-\x1f]')


def safe_filename(name: str | None) -> str:
    """Keeps only the last path component and replaces characters Windows rejects."""
    base = PureWindowsPath(name or "").name
    cleaned = _UNSAFE_CHARS.sub("_", base).strip(" .")
    return cleaned or "upload.pdf"


def summary(services: Services, document_id: str) -> dict:
    meta = services.docstore.metadata(document_id)
    return {
        "document_id": document_id,
        "filename": meta.get("filename"),
        "doc_type": meta.get("doc_type"),
        "date": meta.get("date"),
        "parties": meta.get("parties", []),
        "jurisdiction": meta.get("jurisdiction"),
        "has_pdf": services.docstore.pdf_path(document_id) is not None,
        "indexed": services.indexer.is_indexed(document_id),
    }


@router.get("")
def list_documents(services: ServicesDep) -> list[dict]:
    return [summary(services, document_id) for document_id in services.docstore.list_ids()]


@router.get("/{document_id}")
def get_document(document_id: str, services: ServicesDep) -> dict:
    return summary(services, document_id)


@router.get("/{document_id}/file")
def get_file(document_id: str, services: ServicesDep) -> FileResponse:
    if not services.docstore.exists(document_id):
        raise DocumentNotFound(document_id)
    path = services.docstore.pdf_path(document_id)
    if path is None:
        raise HTTPException(404, "document has no PDF")
    return FileResponse(path, media_type="application/pdf",
                        filename=services.docstore.metadata(document_id).get("filename") or path.name)


@router.post("")
def upload(
    services: ServicesDep,
    file: Annotated[UploadFile, File()],
    doc_type: Annotated[str | None, Form()] = None,
    date: Annotated[str | None, Form()] = None,
    jurisdiction: Annotated[str | None, Form()] = None,
    parties: Annotated[list[str] | None, Form()] = None,
    extract_metadata: Annotated[bool, Form()] = True,
    provider: Annotated[ProviderName | None, Form()] = None,
) -> dict:
    data = file.file.read(MAX_UPLOAD_BYTES + 1)
    if len(data) > MAX_UPLOAD_BYTES:
        raise HTTPException(413, "file is larger than 50 MB")

    given = {
        "doc_type": (doc_type or "").strip() or None,
        "date": (date or "").strip() or None,
        "jurisdiction": (jurisdiction or "").strip() or None,
        "parties": [p.strip() for p in parties or [] if p.strip()] or None,
    }
    warnings: list[str] = []
    extractor = None
    if extract_metadata:
        try:
            llm = services.provider(provider or services.default_provider)
            llm.check_available()
            extractor = MetadataExtractor(llm)
        except (ProviderUnavailableError, ValueError) as exc:
            warnings.append(f"metadata extraction skipped: {exc}")

    # parse_pdf names the document after the file, so keep the uploaded name.
    with tempfile.TemporaryDirectory() as tmp:
        path = Path(tmp) / safe_filename(file.filename)
        path.write_bytes(data)
        result = services.indexer.ingest_pdf(path, given, extract_metadata=extractor)
    return {
        "document_id": result.document_id,
        "chunk_count": result.chunk_count,
        "already_indexed": result.already_indexed,
        "warnings": warnings,
        "document": summary(services, result.document_id),
    }

from dataclasses import asdict
from typing import Literal

from fastapi import APIRouter
from pydantic import BaseModel, Field

from auditor.api.routes.deps import ServicesDep
from auditor.api.schemas import Filters, ProviderName
from auditor.qa.answer import QuestionAnswerer
from auditor.retrieval.config import RetrievalConfig

router = APIRouter(tags=["search"])


class SearchRequest(BaseModel):
    query: str = Field(min_length=1, max_length=2000)
    mode: Literal["bm25", "dense", "hybrid"] = "hybrid"
    rerank: bool = True
    final_k: int = Field(5, ge=1, le=30)
    filters: Filters = Field(default_factory=Filters)


class AskRequest(BaseModel):
    question: str = Field(min_length=1, max_length=2000)
    filters: Filters = Field(default_factory=Filters)
    provider: ProviderName | None = None


@router.post("/search")
def search(body: SearchRequest, services: ServicesDep) -> dict:
    config = RetrievalConfig(mode=body.mode, rerank=body.rerank, final_k=body.final_k)
    result = services.retriever.search(body.query, config, body.filters.to_filter())
    return {
        "query": result.query,
        "config": asdict(result.config),
        "insufficient_evidence": result.insufficient_evidence,
        "chunks": [asdict(chunk) for chunk in result.chunks],
        "parents": [asdict(parent) for parent in result.parents],
    }


@router.post("/ask")
def ask(body: AskRequest, services: ServicesDep) -> dict:
    provider = services.provider(body.provider or services.default_provider)
    provider.check_available()
    answer = QuestionAnswerer(services.retriever, services.docstore, provider).ask(
        body.question, body.filters.to_filter()
    )
    return {
        "question": answer.question,
        "found": answer.found,
        "answer": answer.answer,
        "message": answer.message,
        "citations": [citation.to_dict() for citation in answer.citations],
        "passages": [asdict(passage) for passage in answer.passages],
        "model": answer.model,
        "usage": asdict(answer.usage),
    }

import logging
import threading
import uuid
from concurrent.futures import ThreadPoolExecutor
from dataclasses import asdict, dataclass, replace
from datetime import datetime, timezone

from auditor.audit.runner import Auditor, PreparedAudit

log = logging.getLogger(__name__)


@dataclass
class AuditJob:
    audit_id: str
    document_id: str
    playbook: str
    provider: str
    cache_key: str
    status: str  # running | done | failed
    started_at: str
    error: str | None = None

    def to_dict(self) -> dict:
        data = asdict(self)
        del data["cache_key"]
        return data


class AuditJobs:
    """Runs audits on background threads. Jobs live in memory: finished audits are read
    from the database, and a job that was running when the server stopped is gone."""

    def __init__(self, auditor: Auditor, max_workers: int = 2):
        self._auditor = auditor
        self._pool = ThreadPoolExecutor(max_workers=max_workers, thread_name_prefix="audit")
        self._jobs: dict[str, AuditJob] = {}
        self._lock = threading.Lock()

    def submit(self, prepared: PreparedAudit) -> AuditJob:
        with self._lock:
            for job in self._jobs.values():
                if job.status == "running" and job.cache_key == prepared.cache_key:
                    return replace(job)
            job = AuditJob(
                audit_id=uuid.uuid4().hex,
                document_id=prepared.document_id,
                playbook=prepared.playbook.key,
                provider=prepared.provider.name,
                cache_key=prepared.cache_key,
                status="running",
                started_at=datetime.now(timezone.utc).isoformat(timespec="seconds"),
            )
            self._jobs[job.audit_id] = job
        self._pool.submit(self._run, job.audit_id, prepared)
        return replace(job)

    def _run(self, audit_id: str, prepared: PreparedAudit) -> None:
        try:
            self._auditor.execute(prepared, audit_id)
        except Exception as exc:
            log.exception("audit %s failed", audit_id)
            status, error = "failed", f"{type(exc).__name__}: {exc}"
        else:
            status, error = "done", None
        with self._lock:
            job = self._jobs[audit_id]
            job.status, job.error = status, error

    def get(self, audit_id: str) -> AuditJob | None:
        with self._lock:
            job = self._jobs.get(audit_id)
            return None if job is None else replace(job)

    def shutdown(self) -> None:
        self._pool.shutdown(wait=False, cancel_futures=True)
```

    document_id: str
    playbook: str
    provider: ProviderName | None = None
    force: bool = False


class ReviewRequest(BaseModel):
    decision: Literal["accepted", "rejected"]

@router.post("")
def start_audit(body: AuditRequest, response: Response, services: ServicesDep,
                jobs: JobsDep) -> dict:
    prepared = services.auditor.prepare(body.document_id, body.playbook,
                                        body.provider or services.default_provider)
    if not body.force:
        report = services.auditor.cached(prepared)
        if report is not None:
            return report
    response.status_code = 202
    return jobs.submit(prepared).to_dict()

def list_audits(services: ServicesDep, document_id: str | None = None) -> list[dict]:
    return services.audits.list_audits(document_id)

def get_audit(audit_id: str, response: Response, services: ServicesDep, jobs: JobsDep) -> dict:
    report = services.audits.get_report(audit_id)
    if report is not None:
        return report
    job = jobs.get(audit_id)
    if job is None:
        raise AuditNotFound(audit_id)
    if job.status == "failed":
        raise HTTPException(500, f"audit failed: {job.error}")
    response.status_code = 202
    return job.to_dict()

def get_trace(audit_id: str, services: ServicesDep, jobs: JobsDep) -> list[dict]:
    trace = services.audits.get_trace(audit_id)

import io
from html import escape
from pathlib import Path

import pymupdf

STATUS_TEXT = {
    "uyumlu": "Compliant",
    "uyumsuz": "Non-compliant",
    "eksik": "Missing",
    "belirsiz": "Uncertain",
    "hata": "Error",
}
# Same status colours as the UI; a text label always accompanies them.
STATUS_RGB = {
    "uyumlu": (0.047, 0.639, 0.047),
    "uyumsuz": (0.816, 0.231, 0.231),
    "eksik": (0.925, 0.514, 0.353),
    "belirsiz": (0.980, 0.698, 0.098),
    "hata": (0.537, 0.529, 0.506),
}
_CSS = """
body { font-family: sans-serif; font-size: 10pt; color: #0b0b0b; }
h1 { font-size: 16pt; margin-bottom: 4pt; }
h2 { font-size: 12pt; margin-top: 14pt; margin-bottom: 2pt; }
p { margin-top: 2pt; margin-bottom: 2pt; }
.muted { color: #52514e; }
table { border-collapse: collapse; width: 100%; }
td, th { border: 0.5pt solid #c3c2b7; padding: 3pt; text-align: left; vertical-align: top; }
blockquote { margin-left: 10pt; color: #52514e; }
"""


def _status(value: str) -> str:
    return STATUS_TEXT.get(value, value)


def _finding_html(finding: dict) -> str:
    evidence = finding["evidence"]
    parts = [
        f"<h2>{escape(finding['title'])} ({escape(finding['rule_id'])})</h2>",
        f"<p><b>{escape(_status(finding['status']))}</b> · severity "
        f"{escape(finding['severity'])} · risk {finding['risk']}</p>",
        f"<p class='muted'>Evidence: {evidence['queries_tried']} queries, "
        f"{evidence['sections_examined']} sections examined</p>",
    ]
    if finding.get("reasoning"):
        parts.append(f"<p>{escape(finding['reasoning'])}</p>")
    if finding.get("error"):
        parts.append(f"<p>Error: {escape(finding['error'])}</p>")
    for citation in finding["citations"]:
        page = citation["page"] if citation["page"] is not None else "-"
        parts.append(f"<blockquote>“{escape(citation['quote'])}” "
                     f"(p{page}, {escape(citation['section_path'] or '-')})</blockquote>")
    review = finding.get("review")
    if review:
        parts.append(f"<p>Review: <b>{escape(review['decision'])}</b> {escape(review['note'])}</p>")
    return "\n".join(parts)


def report_html(report: dict) -> str:
    risk = "-" if report["risk_score"] is None else f"{report['risk_score']:.1f}"
    totals = report["totals"]
    rows = "".join(
        f"<tr><td>{escape(f['rule_id'])}</td><td>{escape(_status(f['status']))}</td>"
        f"<td>{escape(f['severity'])}</td><td>{f['risk']}</td></tr>"
        for f in report["findings"]
    )
    findings = "".join(_finding_html(f) for f in report["findings"])
    return (
        "<h1>Compliance audit</h1>"
        f"<p><b>Risk score: {risk}</b> (0 = no risk, 100 = highest)</p>"
        f"<p class='muted'>Playbook {escape(report['playbook'])} · {escape(report['provider'])} "
        f"{escape(report['model'])} · {escape(report['created_at'])} · {totals['duration_s']} s · "
        f"${totals['cost_usd']:.4f}</p>"
        f"<p class='muted'>Audit {escape(report['audit_id'])} · document "
        f"{escape(report['document_id'])}</p>"
        f"<table><tr><th>Rule</th><th>Status</th><th>Severity</th><th>Risk</th></tr>{rows}</table>"
        f"{findings}"
    )


def _render_html(html: str) -> pymupdf.Document:
    buffer = io.BytesIO()
    writer = pymupdf.DocumentWriter(buffer)
    story = pymupdf.Story(html=html, user_css=_CSS)
    mediabox = pymupdf.paper_rect("a4")
    where = mediabox + (50, 50, -50, -50)
    more = True
    while more:
        device = writer.begin_page(mediabox)
        more, _ = story.place(where)
        story.draw(device)
        writer.end_page()
    writer.close()
    return pymupdf.open("pdf", buffer.getvalue())


def report_pdf(report: dict, source_pdf: Path | None) -> bytes:
    out = _render_html(report_html(report))
    if source_pdf is not None:
        offset = out.page_count
        with pymupdf.open(source_pdf) as source:
            out.insert_pdf(source)
            source_pages = source.page_count
        for finding in report["findings"]:
            color = STATUS_RGB.get(finding["status"], STATUS_RGB["hata"])
            label = f"{finding['rule_id']}: {_status(finding['status'])}"
            for citation in finding["citations"]:
                for box in citation["bboxes"]:
                    if not 1 <= box["page"] <= source_pages:
                        continue
                    page = out[offset + box["page"] - 1]
                    annot = page.add_highlight_annot(pymupdf.Rect(box["bbox"]))
                    annot.set_colors(stroke=color)
                    annot.set_info(content=label)
                    annot.update()
    data = out.tobytes(garbage=3, deflate=True)
    out.close()
    return data

```python
"""Benchmark result files. The benchmark scripts write one JSON file per table into
benchmarks/results/; the API reads them and the UI draws tables and charts. The README
table is generated from the same files, so numbers are never typed by hand. The first
column labels the row; column names must not contain "." or "[" (chart field paths)."""

import json
from pathlib import Path

from pydantic import BaseModel, ValidationError


class BenchmarkFile(BaseModel):
    title: str
    created_at: str | None = None
    rows: list[dict[str, str | float | int | None]]
    notes: str = ""


def load_results(directory: Path) -> list[dict]:
    if not directory.is_dir():
        return []
    results = []
    for path in sorted(directory.glob("*.json")):
        try:
            data = BenchmarkFile.model_validate(json.loads(path.read_text(encoding="utf-8")))
        except (json.JSONDecodeError, ValidationError, UnicodeDecodeError) as exc:
            results.append({"file": path.name, "title": path.stem, "created_at": None,
                            "rows": [], "notes": "", "error": f"unreadable result file: {exc}"})
            continue
        results.append({"file": path.name, **data.model_dump(), "error": None})
    return results

```python
"""HTTP client for the auditor API. The UI reaches the backend only through this class."""

import os

import httpx

DEFAULT_URL = "http://localhost:8000"


class ApiError(Exception):
    def __init__(self, status: int, detail: str):
        super().__init__(detail)
        self.status = status  # 0 when the API could not be reached
        self.detail = detail


def _detail(response: httpx.Response) -> str:
    try:
        body = response.json()
    except ValueError:
        return response.text.strip() or response.reason_phrase
    detail = body.get("detail") if isinstance(body, dict) else None
    if isinstance(detail, list):  # FastAPI request validation errors
        return "; ".join(
            f"{'.'.join(str(part) for part in error.get('loc', [])[1:])}: {error.get('msg')}"
            for error in detail
        )
    return str(detail if detail is not None else body)


class ApiClient:
    def __init__(self, http: httpx.Client):
        self._http = http

    @classmethod
    def from_env(cls) -> "ApiClient":
        url = os.environ.get("AUDITOR_API_URL", DEFAULT_URL)
        # Uploads embed a whole contract, so reads may be slow; a down API fails fast.
        return cls(httpx.Client(base_url=url, timeout=httpx.Timeout(300.0, connect=5.0)))

    def _request(self, method: str, path: str, **kwargs) -> httpx.Response:
        try:
            response = self._http.request(method, path, **kwargs)
        except httpx.TransportError as exc:
            raise ApiError(0, f"API unreachable at {self._http.base_url}: {exc}") from exc
        if response.status_code >= 400:
            raise ApiError(response.status_code, _detail(response))
        return response

    @staticmethod
    def _state(response: httpx.Response) -> tuple[str, dict]:
        return ("running" if response.status_code == 202 else "done"), response.json()

    def health(self) -> dict:
        return self._request("GET", "/health").json()

    def documents(self) -> list[dict]:
        return self._request("GET", "/documents").json()

    def document_file(self, document_id: str) -> bytes:
        return self._request("GET", f"/documents/{document_id}/file").content

    def upload(self, filename: str, data: bytes, metadata: dict, extract_metadata: bool = True,
               provider: str | None = None) -> dict:
        form: dict = {key: value for key, value in metadata.items() if value}
        form["extract_metadata"] = "true" if extract_metadata else "false"
        if provider:
            form["provider"] = provider
        files = {"file": (filename, data, "application/pdf")}
        return self._request("POST", "/documents", files=files, data=form).json()

    def search(self, query: str, mode: str = "hybrid", rerank: bool = True, final_k: int = 5,
               filters: dict | None = None) -> dict:
        body = {"query": query, "mode": mode, "rerank": rerank, "final_k": final_k,
                "filters": filters or {}}
        return self._request("POST", "/search", json=body).json()

    def ask(self, question: str, filters: dict | None = None, provider: str | None = None) -> dict:
        body = {"question": question, "filters": filters or {}, "provider": provider}
        return self._request("POST", "/ask", json=body).json()

    def playbooks(self) -> list[dict]:
        return self._request("GET", "/playbooks").json()

    def providers(self) -> dict:
        return self._request("GET", "/providers").json()

    def start_audit(self, document_id: str, playbook: str, provider: str | None = None,
                    force: bool = False) -> tuple[str, dict]:
        body = {"document_id": document_id, "playbook": playbook, "provider": provider,
                "force": force}
        return self._state(self._request("POST", "/audits", json=body))

    def audit(self, audit_id: str) -> tuple[str, dict]:
        return self._state(self._request("GET", f"/audits/{audit_id}"))

    def audits(self, document_id: str | None = None) -> list[dict]:
        params = {"document_id": document_id} if document_id else None
        return self._request("GET", "/audits", params=params).json()

    def trace(self, audit_id: str) -> list[dict]:
        return self._request("GET", f"/audits/{audit_id}/trace").json()

    def review(self, audit_id: str, rule_id: str, decision: str, note: str = "") -> dict:
        body = {"decision": decision, "note": note}
        return self._request("POST", f"/audits/{audit_id}/findings/{rule_id}/review", json=body).json()

    def export(self, audit_id: str, fmt: str) -> bytes:
        return self._request("GET", f"/audits/{audit_id}/export", params={"format": fmt}).content

    def benchmarks(self) -> list[dict]:
        return self._request("GET", "/benchmarks").json()
```

def trace_rows(events: list[dict]) -> list[dict]:
    if not events:
        return []
    starts = [datetime.fromisoformat(event["started_at"]) for event in events]
    origin = min(starts)
    rows = []
    for event, start in zip(events, starts):
        offset = (start - origin).total_seconds()
        rows.append({
            "rule_id": event["rule_id"],
            "node": event["node"],
            "start_s": round(offset, 3),
            "end_s": round(offset + event["duration_s"], 3),
            "duration_s": event["duration_s"],
            "tokens": event["input_tokens"] + event["output_tokens"],
            "cost_usd": event["cost_usd"],
            "models": ", ".join(event["models"]),
            "input": event["input"],
            "output": event["output"],
        })
    return sorted(rows, key=lambda row: (row["rule_id"], row["start_s"]))


def rule_totals(rows: list[dict]) -> list[dict]:
    totals: dict[str, dict] = {}
    for row in rows:
        entry = totals.setdefault(row["rule_id"], {"Rule": row["rule_id"], "Steps": 0,
                                                   "start": row["start_s"], "end": row["end_s"],
                                                   "Tokens": 0, "Cost ($)": 0.0})
        entry["Steps"] += 1
        entry["start"] = min(entry["start"], row["start_s"])
        entry["end"] = max(entry["end"], row["end_s"])
        entry["Tokens"] += row["tokens"]
        entry["Cost ($)"] += row["cost_usd"]
    return [
        {"Rule": e["Rule"], "Steps": e["Steps"], "Seconds": round(e["end"] - e["start"], 3),
         "Tokens": e["Tokens"], "Cost ($)": round(e["Cost ($)"], 4)}
        for e in sorted(totals.values(), key=lambda e: e["Rule"])
    ]


def _is_number(value) -> bool:
    return isinstance(value, (int, float)) and not isinstance(value, bool)


def numeric_columns(rows: list[dict]) -> list[str]:
    columns = list(dict.fromkeys(key for row in rows for key in row))
    numeric = []
    for column in columns:
        values = [row.get(column) for row in rows if row.get(column) is not None]
        if values and all(_is_number(value) for value in values):
            numeric.append(column)
    return numeric


def label_column(rows: list[dict]) -> str | None:
    numeric = set(numeric_columns(rows))
    columns = list(dict.fromkeys(key for row in rows for key in row))
    return next((column for column in columns if column not in numeric), None)

import streamlit as st

import pdfview
from apiclient import ApiClient, ApiError
from viewmodels import document_label


@st.cache_resource
def get_client() -> ApiClient:
    return ApiClient.from_env()


def load_documents(client: ApiClient) -> list[dict]:
    try:
        return client.documents()
    except ApiError as exc:
        st.error(exc.detail)
        st.stop()


def document_picker(docs: list[dict], label: str = "Document") -> dict | None:
    """A document selector whose choice carries over between pages."""
    if not docs:
        st.info("No documents yet. Upload a contract on the Documents page.")
        return None
    by_id = {doc["document_id"]: doc for doc in docs}
    ids = list(by_id)
    current = st.session_state.get("document_id")
    chosen = st.selectbox(label, ids, index=ids.index(current) if current in ids else 0,
                          format_func=lambda document_id: document_label(by_id[document_id]))
    st.session_state["document_id"] = chosen
    return by_id[chosen]


@st.cache_data(show_spinner=False, max_entries=8)
def _pdf_bytes(_client: ApiClient, document_id: str) -> bytes:
    return _client.document_file(document_id)


def show_source(client: ApiClient, doc: dict | None, annotations: list[dict], key: str) -> None:
    if doc is None:
        return
    if not doc["has_pdf"]:
        st.caption("This document was indexed from plain text; there is no PDF to highlight.")
        return
    try:
        data = _pdf_bytes(client, doc["document_id"])
    except ApiError as exc:
        st.error(exc.detail)
        return
    pdfview.show_pdf(data, annotations, key=key)

Kullanıcıya iki seçeneği sun (PDF.js bileşeni vs sayfa görüntüsü); sayfa görüntüsü seçilirse `ui/pdfview.py` şöyle olur:

```python
import pymupdf
import streamlit as st


def _rgb(color: str) -> tuple[float, float, float]:
    value = color.lstrip("#")
    return tuple(int(value[i:i + 2], 16) / 255 for i in (0, 2, 4))


def show_pdf(data: bytes, annotations: list[dict], key: str) -> None:
    """Renders the pages that carry annotations, with the boxes drawn on them."""
    with pymupdf.open("pdf", data) as doc:
        pages = sorted({a["page"] for a in annotations if 1 <= a["page"] <= doc.page_count}) or [1]
        for number in pages:
            page = doc[number - 1]
            for a in annotations:
                if a["page"] == number:
                    rect = pymupdf.Rect(a["x"], a["y"], a["x"] + a["width"], a["y"] + a["height"])
                    color = _rgb(a["color"])
                    page.draw_rect(rect, color=color, fill=color, fill_opacity=0.25, width=1)
            st.image(page.get_pixmap(dpi=110).tobytes("png"), caption=f"Page {number}")
```
